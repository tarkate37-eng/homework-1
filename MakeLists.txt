cmake_minimum_required(VERSION 3.16)
project(hw1_sorting LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include_directories(include)

# 1. Тесты корректности (собираются в Debug с AddressSanitizer)
add_executable(hw1_tests tests/test_correctness.cpp)
if (MSVC)
    target_compile_options(hw1_tests PRIVATE $<$<CONFIG:Debug>:/fsanitize=address>)
else()
    target_compile_options(hw1_tests PRIVATE $<$<CONFIG:Debug>:-fsanitize=address -g>)
    target_link_options(hw1_tests PRIVATE $<$<CONFIG:Debug>:-fsanitize=address>)
endif()

# 2. Бенчмарки скорости (собираются в Release с флагом оптимизации /O2 или -O3, без санитайзеров)
add_executable(hw1_bench bench/benchmark.cpp)
if (MSVC)
    target_compile_options(hw1_bench PRIVATE $<$<CONFIG:Release>:/O2>)
else()
    target_compile_options(hw1_bench PRIVATE $<$<CONFIG:Release>:-O3>)
endif()
